theorem completed_rows (base : Nat → F) (linked : base copy=base 0) : Satisfies (completed base) originalRows := by
  have arithmetic := (whole_segment (F:=F)).complete base linked
  rw [segment_steps (F:=F),segment_rows (F:=F),←completed_materialized] at arithmetic
  have link : completed base copy=completed base 0 := by
    rw [copy_preserved,outside_preserved base 0 (Or.inl (by decide))]
    exact linked
  have copying := TransferPoseidonCall14Program01.copy_satisfied (completed base) link
  rw [call14_copy_binding] at copying

  rcases List.mem_append.mp member with member|member
  · exact arithmetic row member
  · exact copying row member
