    (support : ∀ row∈prior,∀ term∈row.a++row.b,term.1<61630 ∨ 61930≤term.1) : Satisfies (completed base) prior := by
  intro row member
  have agrees (terms : Linear) (included : ∀ term∈terms,term∈row.a++row.b) : eval (completed base) terms=eval base terms := by

  rw [agrees row.a (by intro term present;exact List.mem_append_left _ present),agrees row.b (by intro term present;exact List.mem_append_right _ present)]
