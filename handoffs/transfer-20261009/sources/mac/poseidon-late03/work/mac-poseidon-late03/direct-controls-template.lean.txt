theorem column_rejection (rows : Array Row) (candidate : Parameters Int 6)
    (ports : State Linear 6) (hints : Fin 6 → FifthHint)
    (rejected : checkColumn p copy rows candidate 61 ports shifted transformed hints 0=false) :
    productionCheck rows candidate ports after hints rowMapping=false := by
  apply Bool.eq_false_iff.mpr
  intro accepted
  have fields := checked_fields rows candidate ports after hints rowMapping accepted
  simp only [checkRound,Bool.and_eq_true] at fields
  have acceptedColumn := List.all_eq_true.mp fields.2.1.1 0 (List.mem_finRange 0)
  exact Bool.false_ne_true (rejected.symm.trans acceptedColumn)

theorem mix_rejection (candidate : Parameters Int 6)
    (rejected : decide (canonical p (after 0)=canonical p (mixLinear candidate.mds transformed 0))=false) :
    productionCheck originalRows candidate before after fifthHints rowMapping=false := by
  apply Bool.eq_false_iff.mpr
  intro accepted
  have fields := checked_fields _ _ _ _ _ _ accepted
  have round := fields.2.1
  simp only [checkRound,Bool.and_eq_true] at round
  have acceptedMix := List.all_eq_true.mp round.2 0 (List.mem_finRange 0)
  exact Bool.false_ne_true (rejected.symm.trans acceptedMix)
